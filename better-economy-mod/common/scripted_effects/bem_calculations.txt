bem_add_to_matrix_if_consumed_subp = {
    if = {
        limit = {
            modifier:goods_input_$gj$_add > 0
        }
		owner = { if = {
			limit = {
				AND = { 
					NOT = { has_local_variable = bem_production_matrix_$j$_$i$ }
					has_variable = bem_production_matrix_$j$_$i$
				}
			}
			set_variable = {
				name = bem_production_matrix_$j$_$i$
				value = 0
			}
		} } 
		owner = { if = {
			limit = {
				AND = {
					NOT = { has_local_variable = bem_employment_qt_$j$_$i$ }
					has_variable = bem_v_$i$
				}
			}
			set_variable = {
				name = bem_v_$i$
				value = 0
			}
		}}
        set_local_variable = {
            name = bem_production_matrix_$j$_$i$
            value = modifier:goods_input_$gj$_add
        }
		set_local_variable = {
			name = bem_employment_qt_$j$_$i$
			value = bem_building_total_employment_lw
		}
		owner = {
			if = {
				limit = {
					NOT = {
						has_variable = bem_production_matrix_$j$_$i$
					}
				}
				set_variable = {
					name = bem_production_matrix_$j$_$i$
					value = 0
				}
			}
			if = {
				limit = {
					NOT = {
						has_variable = bem_v_$i$
					}
				}
				set_variable = {
					name = bem_v_$i$
					value = 0
				}
			}
			change_variable = {
				name = bem_production_matrix_$j$_$i$
				add = {
					value = local_var:bem_production_matrix_$j$_$i$
					divide = market.mg:$gi$.market_goods_production
				}
			}
			change_variable = {
				name = bem_v_$i$
				add = {
					value = local_var:bem_employment_qt_$j$_$i$
					divide = market.mg:$gi$.market_goods_production
				}
			}
		}
    }
}


bem_calculate_consumption_of_good_subp = {
	every_scope_building = {
		if = {
			limit = {
				modifier:goods_output_$gi$_add > 0
			}
			bem_add_to_matrix_if_consumed = {
				i = $i$
				gi = $gi$
			}
		}
	}
}

bem_tr_matrix_subp_subp_subp = {
	change_variable = {
		name = bem_tr_matrix_$i$_$j$
		add = {
			value = 0
			if = {
				limit = {
					has_variable = bem_production_matrix_$i$_$k$
					has_variable = bem_production_matrix_$k$_$j$
				}
				add = var:bem_production_matrix_$i$_$k$
				multiply = var:bem_production_matrix_$k$_$j$
			}
		}
	}
}
# Forward substitution: y_i -= var_i_k_tr * y_k, for k < i
bem_forward_sub_subp_subp_subp = {
    if = {
        limit = {
            has_variable = bem_tr_matrix_$i$_$k$
            has_variable = bem_y_$k$
			has_variable = bem_y_$i$
        }
        change_variable = {
            name = bem_y_$i$
            add = {
                value = 0
                add = var:bem_tr_matrix_$i$_$k$
                multiply = var:bem_y_$k$
                multiply = -1
            }
        }
    }
}

bem_clear_subp_subp = {
	if = {
		limit = {
			var:bem_tr_matrix_$i$_$j$ = 0
		}
		remove_variable = bem_tr_matrix_$i$_$j$
	}
}

# Back substitution: x_i -= var_i_k_tr * x_k, for k > i
bem_back_sub_subp_subp_subp = {
    if = {
        limit = {
            has_variable = bem_tr_matrix_$i$_$k$
            has_variable = bem_x_$k$
        }
        change_variable = {
            name = bem_x_$i$
            add = {
                value = 0
                add = var:bem_tr_matrix_$i$_$k$
                multiply = var:bem_x_$k$
                multiply = -1
            }
        }
    }
}

# Back substitution diagonal divide: x_i /= var_i_i_tr
bem_back_sub_divide_subp = {
    change_variable = {
        name = bem_x_$i$
        divide = var:bem_tr_matrix_$i$_$i$
    }
    set_variable = {
        name = bem_$gi$_lw_content
        value = var:bem_x_$i$
    }
	remove_variable = bem_x_$i$
}
